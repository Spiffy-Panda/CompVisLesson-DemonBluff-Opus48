# Runtime + dev dependencies for the Demon Bluff CV pipeline.
#
# Installed into a REPO-LOCAL virtual environment (.venv/, gitignored), created with
# the Python standard library:
#
#     python -m venv .venv
#     .venv/Scripts/python.exe -m pip install -r requirements.txt   # Windows
#     .venv/bin/python        -m pip install -r requirements.txt    # macOS/Linux
#
# Why venv+pip (not uv or conda) for the project itself: it is the zero-install,
# universally-reproducible baseline — every Python ships `venv`, every learner can
# follow along without first installing a tool. uv (speed) and conda (binary/non-Python
# deps) are taught as alternatives in the environment-management lesson module, not
# required to run this code. Versions are pinned for reproducibility.

# --- core array + image IO ---
numpy==2.5.0
pillow==12.2.0

# --- classical computer vision (headless: no GUI deps, server-appropriate) ---
opencv-python-headless==4.13.0.92

# --- REST serving (models loaded once in lifespan; inference in plain def) ---
fastapi==0.138.0
uvicorn[standard]==0.49.0
pydantic==2.13.4
pydantic-settings==2.14.2
python-multipart==0.0.32            # multipart/form-data parsing for frame uploads (UploadFile/Form)

# --- testing + the FastAPI TestClient transport ---
httpx==0.28.1
pytest==9.1.1

# --- model export + serving (ONNX) ---
onnx==1.22.0                        # dev: export the embedding backbone to ONNX + validate it
onnxruntime==1.27.0                 # runtime: serve the small backbone on CPU (no CUDA needed)

# --- dev-time deep-learning stack (GPU, machine-specific — NOT auto-installed by this file) ---
# torch is DEV-time only: export the frozen embedding backbone to ONNX, embed the gallery, and
# (later) train the closed-vocab OCR recognizer. The RUNTIME path is ONNX-on-CPU (above), so torch
# is not required to *serve* the pipeline. The CUDA build is hardware-specific: this project's dev
# box is an NVIDIA Titan Xp (Pascal, sm_61); the cu118 wheels ship Pascal kernels and run under a
# CUDA-13-capable driver. Verified working 2026-06-22 (torch 2.7.1+cu118, torchvision 0.22.1+cu118):
#     .venv/Scripts/python.exe -m pip install torch torchvision --index-url https://download.pytorch.org/whl/cu118
# CPU-only alternative (smaller, no GPU): pip install torch torchvision   # default PyPI build
#
# Permanently avoided:
#   imagehash        — Stage 0 perceptual-hash dedup is a pure-numpy dHash in
#                      src/dbcv/frame_select.py, so this dependency is never needed.
